ll ext_gcd(ll a, ll b, ll &x, ll &y) {
    if(!b) {
        x = 1;
        y = 0;
        return a;
    }

    ll x1, y1;
    ll g = ext_gcd(b, a % b, x1, y1);

    x = y1;
    y = x1 - (a / b) * y1;

    return g;
}

ll mod(ll a, ll m) {
    return (a % m + m) % m;
}

bool crt_merge(ll a, ll m, ll b, ll n, ll &res, ll &lcm) {
    ll x, y;
    ll g = ext_gcd(m, n, x, y);
    ll diff = b - a;

    if(diff % g != 0)
        return false;

    ll ng = n / g;
    ll k = (ll)((i128)(diff / g) * x % ng);
    k = mod(k, ng);

    i128 new_lcm = (i128)m / g * n;
    i128 new_res = (i128)a + (i128)m * k;

    lcm = (ll)new_lcm;
    res = (ll)(new_res % new_lcm);

    if(res < 0)
        res += lcm;

    return true;
}

bool crt(const vector<ll> &a, const vector<ll> &m, ll &res, ll &lcm) {
    res = 0;
    lcm = 1;

    for(int i = 0; i < (int)a.size(); i++) {
        if(m[i] <= 0)
            return false;

        ll next_res, next_lcm;

        if(!crt_merge(res, lcm, mod(a[i], m[i]), m[i],
                      next_res, next_lcm))
            return false;

        res = next_res;
        lcm = next_lcm;
    }

    return true;
}

int gcd(int a, int b) {
  if (!a || !b)
    return a | b;
  unsigned shift = __builtin_ctz(a | b);
  a >>= __builtin_ctz(a);
  do {
    b >>= __builtin_ctz(b);

    b -= a;

#include<bits/stdc++.h>
using namespace std;

const int N = 3e5 + 9;
using ll = long long;

template<typename T = int, int B = 31>
struct Basis {
  T basis[B]; int sz;
  void clear() {
    memset(basis, 0, sizeof basis);
    sz = 0;
  }
  Basis() {
    clear();
  }
  void insert(T x) {
    for (int i = B - 1; i >= 0; i--) {
      if (x >> i & 1) {
        if (basis[i]) x ^= basis[i];
        else {
          basis[i] = x;
          sz++;
          break;
        }
      }
    }
  }
  // is there any subset having xor = x
  bool can(T x) {
    for(int i = B - 1; i >= 0; i--) {
      x = min(x, x ^ basis[i]);
    }
    return x == 0;
  }
  // max subset xor after xoring it with x
  T max_xor(T x = 0) { 
    for(int i = B - 1; i >= 0; i--) {
      x = max(x, x ^ basis[i]);
    }
    return x;
  }
  // kth smallest subset xor, 1st is 0
  T kth(T k) {
    if (k < 1 || k > ((T)1 << sz)) {
      return -1;
    }
    T x = 0;
    T cnt = ((T)1 << sz);
    for (int i = B - 1; i >= 0; i--) {
      if (basis[i]) {
        if (k > (cnt >> 1)) {
          // set the ith bit
          if (!(x >> i & 1)) {
            x ^= basis[i];
          }
          k -= (cnt >> 1);
        } else {
          // unset the ith bit
          if (x >> i & 1) {
            x ^= basis[i];
          }
        }
        cnt >>= 1;
      }
    }
    return x;
  }
  // number of subsets having xor < x
  T count_lt(T x) {
    if (x < 0) {
      return 0;
    }
    T ans = 0;
    T cnt = ((T)1 << sz);
    T mask = 0;
    for (int i = B - 1; i >= 0; i--) {
      // at this stage, all prev > i th bits in mask and x are the same
      if (basis[i]) {
        if (x >> i & 1) {
          ans += (cnt >> 1);
          if (!(mask >> i & 1)) {
            mask ^= basis[i];
          }
        } else {
          if (mask >> i & 1) {
            mask ^= basis[i];
          }
        }
        cnt >>= 1;
      } else {
        if ((x >> i & 1) != (mask >> i & 1)) {
          if (x >> i & 1) {
            return ans + cnt;
          } else {
            return ans; 
          }
        }
      }
    }
    return ans;
  }
};

#include <bits/stdc++.h>
using namespace std;

struct AhoCorasick {
    static constexpr int ALPHA = 26;
    static constexpr char BASE = 'a';

    struct Node {
        array<int, ALPHA> next;
        int link;
        int outputLink;
        vector<int> output;

        // Complejidad: O(ALPHA)
        Node() : link(0), outputLink(-1) {
            next.fill(-1);
        }
    };

    vector<Node> trie;
    vector<int> patternLength;
    bool built;

    // Complejidad: O(ALPHA)
    AhoCorasick() {
        init();
    }

    // Reinicia toda la estructura.
    // Complejidad: O(total de nodos)
    void init() {
        trie.clear();
        trie.emplace_back();
        patternLength.clear();
        built = false;
    }

    // Convierte un caracter valido en su indice.
    // Complejidad: O(1)
    int charId(char c) const {
        return c - BASE;
    }

    // Verifica si el caracter pertenece al alfabeto.
    // Complejidad: O(1)
    bool validChar(char c) const {
        int id = charId(c);
        return 0 <= id && id < ALPHA;
    }

    // Agrega un patron y retorna su identificador.
    // Los identificadores se asignan desde cero.
    // Complejidad: O(longitud del patron)
    int addPattern(const string& pattern) {
        int patternId = (int)patternLength.size();
        patternLength.push_back((int)pattern.size());

        int node = 0;

        for (char c : pattern) {
            int id = charId(c);

            assert(0 <= id && id < ALPHA);

            if (trie[node].next[id] == -1) {
                trie[node].next[id] = (int)trie.size();
                trie.emplace_back();
            }

            node = trie[node].next[id];
        }

        trie[node].output.push_back(patternId);
        built = false;

        return patternId;
    }

    // Construye failure links, output links y transiciones completas.
    // Debe ejecutarse despues de agregar todos los patrones.
    // Complejidad: O(numero de nodos * ALPHA)
    void build() {
        queue<int> q;

        trie[0].link = 0;
        trie[0].outputLink = -1;

        for (int c = 0; c < ALPHA; ++c) {
            int child = trie[0].next[c];

            if (child == -1) {
                trie[0].next[c] = 0;
            } else {
                trie[child].link = 0;
                trie[child].outputLink = trie[0].output.empty() ? -1 : 0;
                q.push(child);
            }
        }

        while (!q.empty()) {
            int node = q.front();
            q.pop();

            for (int c = 0; c < ALPHA; ++c) {
                int child = trie[node].next[c];

                if (child == -1) {
                    trie[node].next[c] = trie[trie[node].link].next[c];
                    continue;
                }

                int failure = trie[trie[node].link].next[c];
                trie[child].link = failure;

                if (!trie[failure].output.empty()) {
                    trie[child].outputLink = failure;
                } else {
                    trie[child].outputLink = trie[failure].outputLink;
                }

                q.push(child);
            }
        }

        built = true;
    }

    // Retorna el siguiente estado al procesar un caracter.
    // Requiere que build haya sido ejecutado.
    // Complejidad: O(1)
    int advance(int state, char c) const {
        assert(built);

        if (!validChar(c)) {
            return 0;
        }

        return trie[state].next[charId(c)];
    }

    // Ejecuta una funcion para cada patron que termina en el estado actual.
    // callback recibe patternId.
    // Complejidad: O(numero de coincidencias reportadas + cadena de outputs)
    template <class Callback>
    void forEachOutput(int state, Callback callback) const {
        for (int node = state; node != -1; node = trie[node].outputLink) {
            for (int patternId : trie[node].output) {
                callback(patternId);
            }
        }
    }

    // Retorna todas las coincidencias como pares:
    // {posicion inicial, patternId}.
    // Complejidad: O(longitud del texto + numero de coincidencias)
    vector<pair<int, int>> findAll(const string& text) const {
        assert(built);

        vector<pair<int, int>> matches;
        int state = 0;

        for (int i = 0; i < (int)text.size(); ++i) {
            state = advance(state, text[i]);

            forEachOutput(state, int patternId {
                int start = i - patternLength[patternId] + 1;
                matches.push_back({start, patternId});
            });
        }

        return matches;
    }

    // Retorna cuantas veces aparece cada patron en el texto.
    // Cuenta apariciones superpuestas.
    // Complejidad: O(longitud del texto + numero de coincidencias)
    vector<long long> countOccurrences(const string& text) const {
        assert(built);

        vector<long long> count(patternLength.size(), 0);
        int state = 0;

        for (char c : text) {
            state = advance(state, c);

            forEachOutput(state, int patternId {
                ++count[patternId];
            });
        }

        return count;
    }

    // Retorna la primera posicion de cada patron.
    // Retorna -1 para patrones que no aparecen.
    // Complejidad: O(longitud del texto + numero de coincidencias)
    vector<int> firstOccurrence(const string& text) const {
        assert(built);

        vector<int> first(patternLength.size(), -1);
        int state = 0;

        for (int i = 0; i < (int)text.size(); ++i) {
            state = advance(state, text[i]);

            forEachOutput(state, int patternId {
                if (first[patternId] == -1) {
                    first[patternId] = i - patternLength[patternId] + 1;
                }
            });
        }

        return first;
    }

    // Verifica si al menos un patron aparece en el texto.
    // Complejidad: O(longitud del texto) en el peor caso
    bool containsAny(const string& text) const {
        assert(built);

        int state = 0;

        for (char c : text) {
            state = advance(state, c);

            if (!trie[state].output.empty() || trie[state].outputLink != -1) {
                return true;
            }
        }

        return false;
    }

    // Verifica si un patron especifico aparece en el texto.
    // Complejidad: O(longitud del texto + numero de coincidencias)
    bool containsPattern(const string& text, int targetId) const {
        assert(built);
        assert(0 <= targetId && targetId < (int)patternLength.size());

        int state = 0;

        for (char c : text) {
            state = advance(state, c);

            bool found = false;

            forEachOutput(state, int patternId {
                if (patternId == targetId) {
                    found = true;
                }
            });

            if (found) {
                return true;
            }
        }

        return false;
    }

    // Retorna la cantidad de nodos del automata.
    // Complejidad: O(1)
    int nodeCount() const {
        return (int)trie.size();
    }

    // Retorna la cantidad de patrones agregados.
    // Complejidad: O(1)
    int patternCount() const {
        return (int)patternLength.size();
    }
};\end{lstlisting}

struct StrHash
{ // Hash polinomial con exponentes decrecientes.
    static constexpr ll ms[] = {1'000'000'007, 1'000'000'403};
    static constexpr ll b = 500'000'000;
    vector<ll> hs[2], bs[2];
    StrHash(string const &s)
    {
        int n = s.length();
        for(int k=0;k<2;k++)
        {
            hs[k].resize(n + 1), bs[k].resize(n + 1, 1);
            for(int i=0;i<n;i++)
            {
                hs[k][i + 1] = (hs[k][i] * b + s[i]) % ms[k];
                bs[k][i + 1] = bs[k][i] * b % ms[k];
            }
        }
    }
    ll get(int idx, int len) const
    { // Hashes en `s[idx, idx+len)`.
        ll h[2];
        for(int k=0;k<2;k++)
        {
            h[k] = hs[k][idx + len] - hs[k][idx] * bs[k][len] % ms[k];
            if (h[k] < 0)
                h[k] += ms[k];
        }
        return (h[0] << 32) | h[1];
    }
};

ll concat_cross_hashes(const StrHash& A, int i1, int len1, const StrHash& B, int i2, int len2)
{
    ll res[2];
    for (int k = 0; k < 2; ++k)
    {
        // hash de substring A[i1..i1+len1-1]
        ll h1 = A.hs[k][i1 + len1] - A.hs[k][i1] * A.bs[k][len1] % A.ms[k];
        if (h1 < 0) h1 += A.ms[k];

        // hash de substring B[i2..i2+len2-1]
        ll h2 = B.hs[k][i2 + len2] - B.hs[k][i2] * B.bs[k][len2] % B.ms[k];
        if (h2 < 0) h2 += B.ms[k];

        // combinacion: h1 * b^len2 + h2
        res[k] = (h1 * A.bs[k][len2] + h2) % A.ms[k];
    }
    return (res[0] << 32) | res[1];
}

        return r;

struct base {
  double a, b;
  base(double a = 0, double b = 0) : a(a), b(b) {}
  const base operator + (const base &c) const {
    return base(a + c.a, b + c.b);
  }
  const base operator - (const base &c) const {
    return base(a - c.a, b - c.b);
  }
  const base operator * (const base &c) const {
    return base(a * c.a - b * c.b, a * c.b + b * c.a);
  }
};
void fft(vector<base> &p, bool inv = 0) {
  int n = p.size(), i = 0;
  for(int j = 1; j < n - 1; ++j) {
    for(int k = n >> 1; k > (i ^= k); k >>= 1);
    if(j < i) swap(p[i], p[j]);
  }
  for(int l = 1, m; (m = l << 1) <= n; l <<= 1) {
    double ang = 2 * PI / m;
    base wn = base(cos(ang), (inv ? 1. : -1.) * sin(ang)), w;
    for(int i = 0, j, k; i < n; i += m) {
      for(w = base(1, 0), j = i, k = i + l; j < k; ++j, w = w * wn) {
        base t = w * p[j + l];
        p[j + l] = p[j] - t;
        p[j] = p[j] + t;
      }
    }
  }
  if(inv) for(int i = 0; i < n; ++i) p[i].a /= n, p[i].b /= n;
}

  stable_sort(sa.begin(), sa.end(), cmp);
  head[0] = col[sa[0]] = cc = 0;
  fore(i, 1, n){
    if(s[sa[i]] != s[sa[i-1]])
      cc++, head[cc] = i;
    col[sa[i]] = cc;
  }
  for(int k = 1; k < n; k *= 2){
    fore(i, 0, n){
      ax = (sa[i] - k + n) % n;
      sa1[head[col[ax]]++] = ax;
    }
    swap(sa, sa1);
    col1[sa[0]] = head[0] = cc = 0;
    fore(i, 1, n){
      if(col[sa[i]] != col[sa[i - 1]] || col[(sa[i] + k) % n] != col[(sa[i - 1] + k) % n])
        cc++, head[cc] = i;
      col1[sa[i]] = cc;
    }
    swap(col, col1); 
    if(col[sa[n - 1]] == n - 1) break;
  }

struct Trie {
    static constexpr int ALPHABET = 26;

    struct Node {
        int next[ALPHABET];
        int prefixCount;
        int wordCount;

        Node() {
            fill(next, next + ALPHABET, -1);
            prefixCount = 0;
            wordCount = 0;
        }
    };

    vector<Node> trie;

    Trie(int maxNodes = 1) {
        trie.reserve(maxNodes);
        trie.emplace_back(); // Nodo 0: raiz
    }

    // Reinicia completamente el Trie
    void clear() {
        trie.clear();
        trie.emplace_back();
    }

    // Crea un nuevo nodo y devuelve su indice
    int newNode() {
        trie.emplace_back();
        return (int)trie.size() - 1;
    }

    // Inserta una palabra
    // Supone que contiene solamente letras entre 'a' y 'z'
    void insert(const string& word) {
        int node = 0;
        trie[node].prefixCount++;

        for (char ch : word) {
            int c = ch - 'a';

            if (trie[node].next[c] == -1) {
                trie[node].next[c] = newNode();
            }

            node = trie[node].next[c];
            trie[node].prefixCount++;
        }

        trie[node].wordCount++;
    }

    // Devuelve el nodo donde termina la cadena
    // Devuelve -1 si la cadena no existe como camino en el Trie
    int findNode(const string& word) const {
        int node = 0;

        for (char ch : word) {
            int c = ch - 'a';

            if (c < 0 || c >= ALPHABET)
                return -1;

            if (trie[node].next[c] == -1)
                return -1;

            node = trie[node].next[c];
        }

        return node;
    }

    // Devuelve cuantas veces se inserto exactamente una palabra
    int countWord(const string& word) const {
        int node = findNode(word);

        if (node == -1)
            return 0;

        return trie[node].wordCount;
    }

    // Indica si una palabra fue insertada al menos una vez
    bool contains(const string& word) const {
        return countWord(word) > 0;
    }

    // Devuelve cuantas palabras insertadas comienzan con prefix
    int countPrefix(const string& prefix) const {
        int node = findNode(prefix);

        if (node == -1)
            return 0;

        return trie[node].prefixCount;
    }

    // Elimina una aparicion de la palabra
    // Devuelve false si la palabra no estaba insertada
    bool erase(const string& word) {
        if (!contains(word))
            return false;

        int node = 0;
        trie[node].prefixCount--;

        for (char ch : word) {
            int c = ch - 'a';
            node = trie[node].next[c];
            trie[node].prefixCount--;
        }

        trie[node].wordCount--;
        return true;
    }

    // Devuelve la cantidad de nodos utilizados
    int size() const {
        return (int)trie.size();
    }
};

    long long s1 = abs(a.cross(b, c));

        if (lexComp(points[i], points[pos]))
            pos = i;

            sgn(seq[0].cross(point)) != sgn(seq[0].cross(seq[n - 1])))

            sgn(seq[n - 1].cross(point)) != sgn(seq[n - 1].cross(seq[0])))

        if (seq[pos].cross(point) >= 0)
            l = mid;
        else
            r = mid;

        used[v] = true;
        for (int u : adj[v]) {
            if (!used[u])
                dfs1(u);
        }
        order.push_back(v);

        comp[v] = cl;
        for (int u : adj_t[v]) {
            if (comp[u] == -1)
                dfs2(u, cl);
        }

        order.clear();
        used.assign(n_vertices, false);
        for (int i = 0; i < n_vertices; ++i) {
            if (!used[i])
                dfs1(i);
        }

        return true;
    }

void bellmanFord()
{
	fill(pot, pot + n, INF);
	pot[S] = 0;
	int qt = 0;
	Q[qt++] = S;
	for (int qh = 0; (qh - qt) % n != 0; qh++)
	{
		int u = Q[qh % n];
		inQueue[u] = false;
		for (int i = 0; i < (int)G[u].size(); i++)
		{
			Edge &e = G[u][i];
			if (e.cap <= e.flow) continue;
			int v = e.to;
			int newDist = pot[u] + e.cost;
			if (pot[v] > newDist)
			{
				pot[v] = newDist;
				if (!inQueue[v])
				{
					Q[qt++ % n] = v;
					inQueue[v] = true;
				}
			}
		}
	}
}

			{

			e.flow += df;
			G[v][e.rev].flow -= df;

	}

    return m * x + c;

struct node {
  Line line;
  node* left = nullptr;
  node* right = nullptr;
  node(Line line) : line(line) {}
  void add_segment(Line nw, int l, int r, int L, int R) {
    if (l > r || r < L || l > R) return;
    int m = (l + 1 == r ? l : (l + r) / 2);
    if (l >= L and r <= R) {
      bool lef = nw.eval(l) < line.eval(l);
      bool mid = nw.eval(m) < line.eval(m);
      if (mid) swap(line, nw);
      if (l == r) return;
      if (lef != mid) {
        if (left == nullptr) left = new node(nw);
        else left -> add_segment(nw, l, m, L, R);
      }
      else {
        if (right == nullptr) right = new node(nw);
        else right -> add_segment(nw, m + 1, r, L, R);
      }
      return;
    }
    if (max(l, L) <= min(m, R)) {
      if (left == nullptr) left = new node({0, inf});
      left -> add_segment(nw, l, m, L, R);
    }
    if (max(m + 1, L) <= min(r, R)) {
      if (right == nullptr) right = new node ({0, inf});
      right -> add_segment(nw, m + 1, r, L, R);
    }
  }
  ll query_segment(ll x, int l, int r, int L, int R) {
    if (l > r || r < L || l > R) return inf;
    int m = (l + 1 == r ? l : (l + r) / 2);
    if (l >= L and r <= R) {
      ll ans = line.eval(x);
      if (l < r) {
        if (x <= m && left != nullptr) ans = min(ans, left -> query_segment(x, l, m, L, R));
        if (x > m && right != nullptr) ans = min(ans, right -> query_segment(x, m + 1, r, L, R));
      }
      return ans;
    }
    ll ans = inf;
    if (max(l, L) <= min(m, R)) {
      if (left == nullptr) left = new node({0, inf});
      ans = min(ans, left -> query_segment(x, l, m, L, R));
    }
    if (max(m + 1, L) <= min(r, R)) {
      if (right == nullptr) right = new node ({0, inf});
      ans = min(ans, right -> query_segment(x, m + 1, r, L, R));
    }
    return ans;
  }
